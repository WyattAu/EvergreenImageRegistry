Quick Start

# Pull a hardened image
docker pull ghcr.io/wyattau/evergreenimageregistry/redis:latest

# Run with security constraints
docker run --rm --read-only --cap-drop ALL \
  --security-opt no-new-privileges \
  -p 6379:6379 \
  ghcr.io/wyattau/evergreenimageregistry/redis:latest

# Browse the full catalog
open https://wyattau.github.io/EvergreenImageRegistry/docs/catalog/
